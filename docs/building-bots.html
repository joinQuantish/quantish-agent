Building Trading Bots

The real power of Quantish Agent is combining coding tools with trading tools to build automated trading systems. This guide walks you through creating your first bot.

// price-monitor.js
// Monitors Trump Bitcoin Reserve market

const QUANTISH_API_KEY = process.env.QUANTISH_API_KEY;
const MCP_URL = 'https://quantish-sdk-production.up.railway.app/mcp';
const TOKEN_ID = '123456789...'; // YES token
const THRESHOLD = 0.05; // 5%

async function callMCP(tool, args) {
  const response = await fetch(`${MCP_URL}/execute`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'x-api-key': QUANTISH_API_KEY
    },
    body: JSON.stringify({
      jsonrpc: '2.0',
      method: 'tools/call',
      params: { name: tool, arguments: args },
      id: Date.now()
    })
  });
  return response.json();
}

async function checkPrice() {
  const result = await callMCP('get_price', { tokenId: TOKEN_ID });
  const price = parseFloat(result.result?.content?.[0]?.text || '0');
  
  console.log(`[${new Date().toISOString()}] Price: ${(price * 100).toFixed(2)}%`);
  
  if (price > THRESHOLD) {
    console.log('🚨 ALERT: Price crossed threshold!');
    // Add notification logic here (Telegram, Discord, etc.)
  }
}

// Check every 60 seconds
setInterval(checkPrice, 60000);
checkPrice();

Bot Architecture Patterns

Pattern 1: Simple Polling

// Poll every N seconds
setInterval(async () => {
  const positions = await getPositions();
  for (const pos of positions) {
    if (shouldSell(pos)) {
      await placeOrder({ side: 'SELL', ... });
    }
  }
}, 30000);

Pattern 2: Event-Driven

// React to price changes
const priceCache = new Map();

async function onPriceUpdate(tokenId, newPrice) {
  const oldPrice = priceCache.get(tokenId);
  priceCache.set(tokenId, newPrice);
  
  if (oldPrice && significantChange(oldPrice, newPrice)) {
    await evaluateTrade(tokenId, newPrice);
  }
}

Pattern 3: State Machine

const states = {
  IDLE: 'idle',
  WATCHING: 'watching',
  ENTERING: 'entering',
  IN_POSITION: 'in_position',
  EXITING: 'exiting'
};

let currentState = states.IDLE;

async function transition(event) {
  switch (currentState) {
    case states.IDLE:
      if (event.type === 'OPPORTUNITY') {
        currentState = states.ENTERING;
        await placeEntry(event.market);
      }
      break;
    // ... more state transitions
  }
}

Using the MCP API Directly

Your bots can call the Quantish MCP API directly:

const MCP_URL = process.env.QUANTISH_MCP_URL;
const API_KEY = process.env.QUANTISH_API_KEY;

async function callTool(name, args) {
  const response = await fetch(`${MCP_URL}/execute`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'x-api-key': API_KEY
    },
    body: JSON.stringify({
      jsonrpc: '2.0',
      method: 'tools/call',
      params: { name, arguments: args },
      id: Date.now()
    })
  });
  
  const data = await response.json();
  if (data.error) throw new Error(data.error.message);
  
  // Parse MCP response format
  const content = data.result?.content?.[0];
  if (content?.type === 'text') {
    return JSON.parse(content.text);
  }
  return data.result;
}

// Example usage
const balances = await callTool('get_balances', {});
const positions = await callTool('get_positions', {});
await callTool('place_order', {
  conditionId: '0x...',
  tokenId: '123...',
  side: 'BUY',
  price: 0.55,
  size: 100
});

Best Practices

💾 Log Everything

Log all trades, prices, and decisions. You'll need this for debugging and analysis.

🔒 Error Handling

Wrap all API calls in try/catch. Network errors happen - your bot should survive them.

⏱️ Rate Limiting

Don't poll too frequently. 30-60 second intervals are usually sufficient.

🧪 Paper Trading

Test your logic without real money first. Log what trades "would" happen.
